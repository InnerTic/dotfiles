# =============================================================================
# LLAMA.CPP — DYNAMIC MODEL LOADER (recommended)
# =============================================================================
# Interactive model selector — scans /home/ken/Downloads/llm_models/ for .gguf files
# and lets you pick one to serve on port 8080.
# Path: ~/.local/bin/llama-loader
# Also updates opencode config automatically on model switch.
llama-loader                    # or just: llm

# Kill running llama-server
kill-llama                      # or just: llmk

# Check which model is currently running on port 8080
curl -s http://127.0.0.1:8080/v1/models | jq -r .data[].id
# Alias: llmcheck

# =============================================================================
# LLAMA.CPP — MANUAL START (when you need specific flags)
# =============================================================================
# Hardware: NVIDIA RTX 3060 12GB VRAM, -ngl 35 for 7B-8B models
# Binary path: /home/ken/llama.cpp/build/bin/llama-server (CUDA build)
# Port: 8080 | Context: 16384

# Example — start Qwen3-VL-8B manually with different context:
/home/ken/llama.cpp/build/bin/llama-server \
  -m /home/ken/Downloads/llm_models/Qwen3-VL-8B-Instruct-abliterated-v2.Q5_K_M.gguf \
  -c 8192 \
  -ngl 35 \
  --host 127.0.0.1 \
  --port 8080 &

# Start llama server with custom params (via script)
~/.openclaw/workspace/scripts/llama-start.sh
# Alias: llmstart
